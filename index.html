-- User নিজের Profile দেখতে পারবে DROP POLICY IF EXISTS own_profile_read ON public.profiles; CREATE POLICY own_profile_read ON public.profiles FOR SELECT TO authenticated USING (id = auth.uid()); -- User নিজের Wallet দেখতে পারবে DROP POLICY IF EXISTS own_wallet_read ON public.wallets; CREATE POLICY own_wallet_read ON public.wallets FOR SELECT TO authenticated USING (user_id = auth.uid()); -- Admin সবকিছু পরিচালনা করতে পারবে DO $$ DECLARE tbl TEXT; BEGIN FOREACH tbl IN ARRAY ARRAY[ 'profiles', 'wallets', 'deposits', 'withdrawals', 'ledger' ] LOOP EXECUTE format( 'DROP POLICY IF EXISTS admin_full_access ON public.%I', tbl ); EXECUTE format( 'CREATE POLICY admin_full_access ON public.%I FOR ALL TO authenticated USING (public.is_admin()) WITH CHECK (public.is_admin())', tbl ); END LOOP; END; $$; GRANT SELECT ON public.profiles, public.wallets TO authenticated;
